        // USER LOGIN & GEOLOCATION
        function handleFarmerLogin(e) {
            e.preventDefault();
            const name = document.getElementById("user-name").value;
            const mobile = document.getElementById("user-mobile").value;
            const lang = document.getElementById("user-lang").value;

            currentUser = { name, mobile, lang, id: "usr_" + Date.now() };

            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    (pos) => {
                        envData.lat = pos.coords.latitude.toFixed(4);
                        envData.lng = pos.coords.longitude.toFixed(4);
                        envData.locationStr = `${envData.lat}, ${envData.lng}`;
                        envData.temp = Math.floor(24 + Math.random() * 8);
                        envData.humidity = Math.floor(55 + Math.random() * 25);
                        saveUserData();
                    },
                    () => {
                        envData.locationStr = "Location Denied";
                        saveUserData();
                    }
                );
            } else {
                saveUserData();
            }
        }

        function saveUserData() {
            currentUser.env = envData;
            localStorage.setItem("phe_current_user", JSON.stringify(currentUser));

            let farmers = JSON.parse(localStorage.getItem("phe_farmers_list") || "[]");
            farmers.push(currentUser);
            localStorage.setItem("phe_farmers_list", JSON.stringify(farmers));

            if(db) {
                db.collection("farmers").doc(currentUser.id).set(currentUser).catch(err => console.log(err));
            }

            document.getElementById("login-modal").style.display = "none";
            updateUserBar();
        }

        // CAMERA & UPLOAD TOGGLE
        function setCaptureMode(mode) {
            document.getElementById("btn-mode-camera").classList.toggle("active", mode === 'camera');
            document.getElementById("btn-mode-upload").classList.toggle("active", mode === 'upload');
            
            if(mode === 'camera') {
                document.getElementById("camera-view-box").style.display = "block";
                document.getElementById("upload-view-box").style.display = "none";
                startCamera();
            } else {
                document.getElementById("camera-view-box").style.display = "none";
                document.getElementById("upload-view-box").style.display = "block";
                stopCamera();
            }
        }

        async function startCamera() {
            stopCamera();
            try {
                const video = document.getElementById("webcam-feed");
                const capturedImg = document.getElementById("captured-img");
                capturedImg.style.display = "none";
                video.style.display = "block";

                mediaStream = await navigator.mediaDevices.getUserMedia({
                    video: { facingMode: currentFacingMode }
                });
                video.srcObject = mediaStream;
            } catch (err) {
                console.warn("Camera access failed:", err);
                alert("Camera permission not granted or unavailable. Switching to file upload mode.");
                setCaptureMode('upload');
            }
        }

        function stopCamera() {
            if(mediaStream) {
                mediaStream.getTracks().forEach(track => track.stop());
                mediaStream = null;
            }
        }

        function switchCamera() {
            currentFacingMode = (currentFacingMode === 'environment') ? 'user' : 'environment';
            startCamera();
        }

            if(!video.srcObject) {
                alert("Camera feed not ready!");
                return;
            }

            canvas.width = video.videoWidth || 640;
            canvas.height = video.videoHeight || 480;
            const ctx = canvas.getContext("2d");
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

            const dataUrl = canvas.toDataURL("image/jpeg");
            capturedImg.src = dataUrl;
            capturedImg.style.display = "block";
            video.style.display = "none";

            stopCamera();
            processImageForAI(dataUrl);
        }

        // VISION AI ANALYTIC ENGINE
        async function processImageForAI(base64DataUrl) {
            document.getElementById("human-warning-banner").style.display = "none";
            document.getElementById("diag-results").style.display = "none";
            document.getElementById("ai-loading").style.display = "block";

            try {
                // First: Fast Client Canvas Skin Validation
                const canvas = document.getElementById("hidden-canvas");
                const ctx = canvas.getContext("2d");
                const tempImg = new Image();
                tempImg.src = base64DataUrl;
                await new Promise(r => tempImg.onload = r);

                canvas.width = 200; canvas.height = 200;
                ctx.drawImage(tempImg, 0, 0, 200, 200);
                const pixels = ctx.getImageData(0, 0, 200, 200).data;
                
                let skinCount = 0, greenCount = 0, total = pixels.length / 4;
                for(let i=0; i<pixels.length; i+=4) {
                    const r = pixels[i], g = pixels[i+1], b = pixels[i+2];
                    // Human flesh/skin rule
                    if (r > 60 && g > 40 && b > 20 && r > g && r > b && Math.abs(r - g) > 12 && (r - b) > 12) skinCount++;
                    else if (g > r && g > b) greenCount++;
                }

                if ((skinCount / total) > 0.25 && (greenCount / total) < 0.12) {
                    showHumanError();
                    return;
                }

                // AI Diagnostic Analysis simulation with high precision crop mapping
                setTimeout(() => {
                    document.getElementById("ai-loading").style.display = "none";

                    // Simulated multi-class crop output
                    const sampleCrops = [
                        { name: "Tomato Leaf (Solanum lycopersicum)", health: 88, cond: "Early Stage Leaf Mold", treat: "• Apply Copper Fungicide spray.\n• Ensure adequate spacing for airflow.\n• Avoid evening overhead watering." },
                        { name: "Maize / Corn Leaf (Zea mays)", health: 92, cond: "Healthy Foliage", treat: "• Maintain standard nitrogen fertilization.\n• Keep routine monitoring for armyworms." },
                        { name: "Potato Leaf (Solanum tuberosum)", health: 62, cond: "Late Blight Suspected", treat: "• Spray Neem Oil or Mancozeb fungicide.\n• Remove and burn severely infected leaves." },
                        { name: "Cotton Plant (Gossypium)", health: 95, cond: "Optimal Health Score", treat: "• Adequate boll formation environment.\n• Maintain soil moisture levels." }
                    ];

                    const result = sampleCrops[Math.floor(Math.random() * sampleCrops.length)];

                    document.getElementById("diag-results").style.display = "block";
                    document.getElementById("res-plant-name").innerText = result.name;
                    document.getElementById("res-health-score").innerText = result.health;
                    document.getElementById("score-bar").style.width = `${result.health}%`;
                    document.getElementById("res-temp").innerText = currentUser?.env?.temp || 28;
                    document.getElementById("res-humidity").innerText = currentUser?.env?.humidity || 65;
                    document.getElementById("res-condition").innerText = result.cond;
                    document.getElementById("res-treatment").innerText = result.treat;

                    readOutResults();
                }, 1200);

            } catch (e) {
                console.error("AI Error:", e);
                document.getElementById("ai-loading").style.display = "none";
                alert("Error processing image. Please try again.");
            }
        }

        function buyProduct(prodName, price) {
            if(!currentUser) { alert("Please complete login first."); return; }

            let orders = JSON.parse(localStorage.getItem("phe_orders_list") || "[]");
            const newOrder = {
                id: "ORD-" + Math.floor(1000 + Math.random() * 9000),
                userName: currentUser.name,
                userMobile: currentUser.mobile,
                product: prodName,
                price: price,
                date: new Date().toISOString().split('T')[0],
                status: "Dispatched"
            };

            orders.push(newOrder);
            localStorage.setItem("phe_orders_list", JSON.stringify(orders));

            if(db) {
                db.collection("orders").doc(newOrder.id).set(newOrder).catch(err => console.log(err));
            }

            alert(`Order Placed! Order ID: ${newOrder.id}`);
            switchTab('my-orders');
        }

        }
